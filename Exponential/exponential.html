Exponential Search

function exponentialSearch(arr, target) {
  if (arr[0] === target) return 0;
  let i = 1;
  while (i < arr.length && arr[i] < target) {
    i = i * 2;
  }
  let left = Math.floor(i / 2);
  let right = Math.min(i, arr.length - 1);
  return binarySearch(arr, target, left, right);
}

function binarySearch(arr, target, left, right) {
  while (left <= right) {
    let mid = Math.floor((left + right) / 2);
    if (arr[mid] === target) return mid;
    if (arr[mid] < target) left = mid + 1;
    else right = mid - 1;
  }
  return -1;
}

About Exponential Search

Exponential Search first finds a range where the target may exist by doubling the index, then performs Binary Search within that range. Ideal for unbounded or infinite sorted arrays.
