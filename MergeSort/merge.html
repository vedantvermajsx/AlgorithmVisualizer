Merge Sort

function mergeSort(arr, left, right) {
  if (left < right) {
    let mid = Math.floor((left + right) / 2);
    mergeSort(arr, left, mid);
    mergeSort(arr, mid + 1, right);
    merge(arr, left, mid, right);
  }
}
function merge(arr, left, mid, right) {
  let L = arr.slice(left, mid + 1);
  let R = arr.slice(mid + 1, right + 1);
  let i = 0, j = 0, k = left;
  while (i < L.length && j < R.length)
    arr[k++] = L[i] <= R[j] ? L[i++] : R[j++];
}

About Merge Sort

Merge Sort is a stable divide-and-conquer algorithm. It divides the array into halves, recursively sorts each half, then merges them back together in sorted order.
